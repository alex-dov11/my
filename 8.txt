<?php

// Функция для отправки JSON-ответа
function sendResponse($data, $status = 200) {
    http_response_code($status);
    echo json_encode($data);
    exit;
}

// Функция для валидации (переиспользуем)
function validateApplicationData($data, &$errors) {
    $valid_languages = ['Pascal', 'C', 'C++', 'JavaScript', 'PHP', 'Python', 'Java', 'Haskell', 'Clojure', 'Prolog', 'Scala', 'Go'];
    
    $full_name = trim($data['full_name'] ?? '');
    $phone = trim($data['phone'] ?? '');
    $email = trim($data['email'] ?? '');
    $birthdate = $data['birthdate'] ?? '';
    $gender = $data['gender'] ?? '';
    $languages = $data['languages'] ?? [];
    $biography = trim($data['biography'] ?? '');
    $contract_accepted = isset($data['contract_accepted']) ? 1 : 0;
    
    if (empty($full_name)) {
        $errors['full_name'] = 'ФИО обязательно';
    } elseif (strlen($full_name) > 150) {
        $errors['full_name'] = 'ФИО не более 150 символов';
    } elseif (!preg_match('/^[a-zA-Zа-яА-ЯёЁ\s\-]+$/u', $full_name)) {
        $errors['full_name'] = 'Только буквы, пробелы, дефисы';
    }
    
    if (empty($phone)) {
        $errors['phone'] = 'Телефон обязателен';
    } elseif (!preg_match('/^[\+\d\s\(\)\-]{10,20}$/', $phone)) {
        $errors['phone'] = 'Некорректный телефон';
    }
    
    if (empty($email)) {
        $errors['email'] = 'Email обязателен';
    } elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $errors['email'] = 'Некорректный email';
    }
    
    if (empty($birthdate)) {
        $errors['birthdate'] = 'Дата рождения обязательна';
    } elseif (!preg_match('/^\d{4}-\d{2}-\d{2}$/', $birthdate)) {
        $errors['birthdate'] = 'Формат ГГГГ-ММ-ДД';
    } else {
        $ts = strtotime($birthdate);
        if ($ts < strtotime('1900-01-01') || $ts > time()) {
            $errors['birthdate'] = 'Дата от 1900 до сегодня';
        }
    }
    
    if (!in_array($gender, ['male', 'female', 'other'])) {
        $errors['gender'] = 'Выберите пол';
    }
    
    if (empty($languages)) {
        $errors['languages'] = 'Выберите хотя бы один язык';
    } else {
        foreach ($languages as $lang) {
            if (!in_array($lang, $valid_languages)) {
                $errors['languages'] = "Недопустимый язык: $lang";
                break;
            }
        }
    }
    
    if (strlen($biography) > 5000) {
        $errors['biography'] = 'Биография не больше 5000 символов';
    }
    
    if (!$contract_accepted) {
        $errors['contract_accepted'] = 'Подтвердите контракт';
    }
    
    return [
        'full_name' => $full_name,
        'phone' => $phone,
        'email' => $email,
        'birthdate' => $birthdate,
        'gender' => $gender,
        'languages' => $languages,
        'biography' => $biography,
        'contract_accepted' => $contract_accepted
    ];
}

function generateLogin($app_id) {
    return 'user_' . $app_id . '_' . bin2hex(random_bytes(4));
}

function generatePassword($len = 10) {
    return substr(str_shuffle('abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'), 0, $len);
}

// --- Роутинг ---
try {
    $pdo = new PDO("mysql:host=$host;dbname=$dbname;charset=utf8mb4", $username, $password);
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
    
    // POST — создание новой заявки
    if ($method === 'POST') {
        $errors = [];
        $validated = validateApplicationData($input, $errors);
        
        if (!empty($errors)) {
            sendResponse(['success' => false, 'errors' => $errors], 400);
        }
        
        $pdo->beginTransaction();
        
        // Вставка в application
        $stmt = $pdo->prepare("
            INSERT INTO application (full_name, phone, email, birthdate, gender, biography, contract_accepted) 
            VALUES (?, ?, ?, ?, ?, ?, ?)
        ");
        $stmt->execute([
            $validated['full_name'], $validated['phone'], $validated['email'],
            $validated['birthdate'], $validated['gender'], $validated['biography'],
            $validated['contract_accepted']
        ]);
        $app_id = $pdo->lastInsertId();
        
        // Вставка языков
        $lang_stmt = $pdo->prepare("SELECT id FROM programming_language WHERE name = ?");
        $insert_stmt = $pdo->prepare("INSERT INTO application_language (application_id, language_id) VALUES (?, ?)");
        foreach ($validated['languages'] as $lang) {
            $lang_stmt->execute([$lang]);
            $lang_id = $lang_stmt->fetchColumn();
            if ($lang_id) $insert_stmt->execute([$app_id, $lang_id]);
        }
        
        // Генерация логина и пароля
        $login = generateLogin($app_id);
        $plain_password = generatePassword();
        $hash = password_hash($plain_password, PASSWORD_DEFAULT);
        $stmt = $pdo->prepare("INSERT INTO `user` (login, password_hash, application_id) VALUES (?, ?, ?)");
        $stmt->execute([$login, $hash, $app_id]);
        
        $pdo->commit();
        
        sendResponse([
            'success' => true,
            'message' => 'Данные сохранены!',
            'login' => $login,
            'password' => $plain_password,
            'profile_url' => "http://{$_SERVER['HTTP_HOST']}/lab8/profile.php?id=$app_id"
        ]);
    }
    
    // PUT — обновление существующей заявки (нужна авторизация)
    if ($method === 'PUT') {
        if (!$id) {
            sendResponse(['success' => false, 'error' => 'ID не указан'], 400);
        }
        
        // Проверка авторизации через HTTP Basic Auth
        $auth_user = $_SERVER['PHP_AUTH_USER'] ?? '';
        $auth_pass = $_SERVER['PHP_AUTH_PW'] ?? '';
        
        if (!$auth_user || !$auth_pass) {
            header('WWW-Authenticate: Basic realm="API"');
            sendResponse(['success' => false, 'error' => 'Требуется авторизация'], 401);
        }
        
        // Проверяем пользователя в БД
        $stmt = $pdo->prepare("SELECT u.*, a.id as app_id FROM `user` u JOIN application a ON u.application_id = a.id WHERE u.login = ?");
        $stmt->execute([$auth_user]);
        $user = $stmt->fetch();
        
        if (!$user || !password_verify($auth_pass, $user['password_hash'])) {
            sendResponse(['success' => false, 'error' => 'Неверный логин или пароль'], 401);
        }
        
        // Проверяем, что пользователь редактирует свою заявку
        if ($user['application_id'] != $id) {
            sendResponse(['success' => false, 'error' => 'Нет доступа к этой заявке'], 403);
        }
        
        $errors = [];
        $validated = validateApplicationData($input, $errors);
        
        if (!empty($errors)) {
            sendResponse(['success' => false, 'errors' => $errors], 400);
        }
        
        // Обновляем заявку
        $stmt = $pdo->prepare("
            UPDATE application SET 
            full_name = ?, phone = ?, email = ?, birthdate = ?, 
            gender = ?, biography = ?, contract_accepted = ?
            WHERE id = ?
        ");
        $stmt->execute([
            $validated['full_name'], $validated['phone'], $validated['email'],
            $validated['birthdate'], $validated['gender'], $validated['biography'],
            $validated['contract_accepted'], $id
        ]);
        
        // Обновляем языки
        $pdo->prepare("DELETE FROM application_language WHERE application_id = ?")->execute([$id]);
        $lang_stmt = $pdo->prepare("SELECT id FROM programming_language WHERE name = ?");
        $insert_stmt = $pdo->prepare("INSERT INTO application_language (application_id, language_id) VALUES (?, ?)");
        foreach ($validated['languages'] as $lang) {
            $lang_stmt->execute([$lang]);
            $lang_id = $lang_stmt->fetchColumn();
            if ($lang_id) $insert_stmt->execute([$id, $lang_id]);
        }
        
        sendResponse([
            'success' => true,
            'message' => 'Данные обновлены!'
        ]);
    }
    
    // GET — получение данных заявки
    if ($method === 'GET' && $id) {
        $stmt = $pdo->prepare("
            SELECT a.*, GROUP_CONCAT(pl.name SEPARATOR ',') as languages
            FROM application a
            LEFT JOIN application_language al ON a.id = al.application_id
            LEFT JOIN programming_language pl ON al.language_id = pl.id
            WHERE a.id = ?
            GROUP BY a.id
        ");
        $stmt->execute([$id]);
        $app = $stmt->fetch(PDO::FETCH_ASSOC);
        
        if (!$app) {
            sendResponse(['success' => false, 'error' => 'Заявка не найдена'], 404);
        }
        
        // Получаем логин пользователя
        $stmt = $pdo->prepare("SELECT login FROM `user` WHERE application_id = ?");
        $stmt->execute([$id]);
        $user = $stmt->fetch();
        
        sendResponse([
            'success' => true,
            'data' => $app,
            'login' => $user['login'] ?? null
        ]);
    }
    
    sendResponse(['success' => false, 'error' => 'Метод не поддерживается'], 405);
    
} catch (PDOException $e) {
    sendResponse(['success' => false, 'error' => 'Ошибка БД: ' . $e->getMessage()], 500);
}

document.addEventListener('DOMContentLoaded', function() {
    const form = document.getElementById('application-form');
    if (!form) return;
    
    const apiUrl = form.dataset.apiUrl || '/lab8/api.php';
    const isLoggedIn = form.dataset.isLoggedIn === 'true';
    const applicationId = form.dataset.applicationId || null;
    
    form.addEventListener('submit', async function(e) {
        e.preventDefault();
        
        // Собираем данные формы
        const formData = new FormData(form);
        const data = {};
        
        // Обработка multiple select (языки)
        const languages = [];
        formData.forEach((value, key) => {
            if (key === 'languages[]') {
                languages.push(value);
            } else if (key === 'languages') {
                languages.push(value);
            } else {
                data[key] = value;
            }
        });
        data.languages = languages;
        
        // Определяем метод и URL
        let method = 'POST';
        let url = apiUrl;
        
        if (isLoggedIn && applicationId) {
            method = 'PUT';
            url = `${apiUrl}?id=${applicationId}`;
        }
        
        // Отправляем запрос
        try {
            const response = await fetch(url, {
                method: method,
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify(data)
            });
            
            const result = await response.json();
            
            if (result.success) {
                // Показываем сообщение об успехе
                const messageDiv = document.getElementById('api-message') || document.createElement('div');
                messageDiv.id = 'api-message';
                messageDiv.className = 'success-message';
                messageDiv.innerHTML = `<strong>✅ ${result.message}</strong>`;
                
                if (result.login && result.password) {
                    messageDiv.innerHTML += `<br><strong>Логин:</strong> ${result.login}<br>
                                            <strong>Пароль:</strong> ${result.password}<br>
                                            <a href="${result.profile_url}" target="_blank">Перейти в профиль</a>`;
                }
                
                if (!document.getElementById('api-message')) {
                    form.insertAdjacentElement('beforebegin', messageDiv);
                }
                
                // Очищаем старые ошибки
                document.querySelectorAll('.field-error').forEach(el => el.remove());
                document.querySelectorAll('.error').forEach(el => el.classList.remove('error'));
                
                // Если это PUT — просто обновляем сообщение
                if (method === 'PUT') {
                    // Можно ничего не делать, данные уже обновлены
                }
            } else {
                // Показываем ошибки
                const errors = result.errors || { global: [result.error] };
                
                // Очищаем старые ошибки
                document.querySelectorAll('.field-error').forEach(el => el.remove());
                document.querySelectorAll('.error').forEach(el => el.classList.remove('error'));
                
                // Показываем ошибки рядом с полями
                for (const [field, errorMsg] of Object.entries(errors)) {
                    if (field === 'global') continue;
                    
                    const input = form.querySelector(`[name="${field}"]`) || 
                                  form.querySelector(`[name="${field}[]"]`);
                    if (input) {
                        input.classList.add('error');
                        const errorDiv = document.createElement('div');
                        errorDiv.className = 'field-error';
                        errorDiv.textContent = errorMsg;
                        input.parentNode.insertBefore(errorDiv, input.nextSibling);
                    }
                }
                
                // Глобальные ошибки
                const globalErrors = errors.global || [result.error];
                if (globalErrors.length) {
                    const messageDiv = document.getElementById('api-message') || document.createElement('div');
                    messageDiv.id = 'api-message';
                    messageDiv.className = 'global-error-message';
                    messageDiv.innerHTML = `<strong>⚠️ Ошибки:</strong><ul>${globalErrors.map(e => `<li>${e}</li>`).join('')}</ul>`;
                    if (!document.getElementById('api-message')) {
                        form.insertAdjacentElement('beforebegin', messageDiv);
                    }
                }
            }
        } catch (error) {
            console.error('Ошибка:', error);
            const messageDiv = document.getElementById('api-message') || document.createElement('div');
            messageDiv.id = 'api-message';
            messageDiv.className = 'global-error-message';
            messageDiv.innerHTML = '<strong>⚠️ Ошибка соединения с сервером</strong>';
            if (!document.getElementById('api-message')) {
                form.insertAdjacentElement('beforebegin', messageDiv);
            }
        }
    });
});
