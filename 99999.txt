function validateData($data, &$errors) {
    $valid_languages = ['Pascal', 'C', 'C++', 'JavaScript', 'PHP', 'Python', 'Java', 'Haskell', 'Clojure', 'Prolog', 'Scala', 'Go'];
    
    $full_name = trim($data['full_name'] ?? '');
    $phone = trim($data['phone'] ?? '');
    $email = trim($data['email'] ?? '');
    $birthdate = $data['birthdate'] ?? '';
    $gender = $data['gender'] ?? '';
    $languages = $data['languages'] ?? [];
    $biography = trim($data['biography'] ?? '');
    $contract_accepted = isset($data['contract_accepted']) ? 1 : 0;
    
    if (empty($full_name)) {
        $errors['full_name'] = 'ФИО обязательно';
    } elseif (strlen($full_name) > 150) {
        $errors['full_name'] = 'ФИО не более 150 символов';
    } elseif (!preg_match('/^[a-zA-Zа-яА-ЯёЁ\s\-]+$/u', $full_name)) {
        $errors['full_name'] = 'Только буквы, пробелы, дефисы';
    }
    
    if (empty($phone)) {
        $errors['phone'] = 'Телефон обязателен';
    } elseif (!preg_match('/^[\+\d\s\(\)\-]{5,25}$/', $phone)) {
        $errors['phone'] = 'Некорректный телефон';
    }
    
    if (empty($email)) {
        $errors['email'] = 'Email обязателен';
    } elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $errors['email'] = 'Некорректный email';
    }
    
    if (empty($birthdate)) {
        $errors['birthdate'] = 'Дата рождения обязательна';
    } elseif (!preg_match('/^\d{4}-\d{2}-\d{2}$/', $birthdate)) {
        $errors['birthdate'] = 'Формат ГГГГ-ММ-ДД';
    } else {
        $ts = strtotime($birthdate);
        if ($ts < strtotime('1900-01-01') || $ts > time()) {
            $errors['birthdate'] = 'Дата от 1900 до сегодня';
        }
    }
    
    if (!in_array($gender, ['male', 'female', 'other'])) {
        $errors['gender'] = 'Выберите пол';
    }
    
    if (empty($languages)) {
        $errors['languages'] = 'Выберите хотя бы один язык';
    } else {
        foreach ($languages as $lang) {
            if (!in_array($lang, $valid_languages)) {
                $errors['languages'] = "Недопустимый язык: $lang";
                break;
            }
        }
    }
    
    if (strlen($biography) > 5000) {
        $errors['biography'] = 'Биография не больше 5000 символов';
    }
    
    if (!$contract_accepted) {
        $errors['contract_accepted'] = 'Подтвердите контракт';
    }
    
    return compact('full_name', 'phone', 'email', 'birthdate', 'gender', 'languages', 'biography', 'contract_accepted');
}

function sendResponse($data, $status = 200) {
    http_response_code($status);
    echo json_encode($data, JSON_UNESCAPED_UNICODE);
    exit;
}

try {

        if (!$id) {

        $auth_user = $_SERVER['PHP_AUTH_USER'] ?? '';
        $auth_pass = $_SERVER['PHP_AUTH_PW'] ?? '';
        
        if (!$auth_user || !$auth_pass) {
            sendResponse(['success' => false, 'error' => 'Требуется авторизация'], 401);
        }
        
        $stmt = $pdo->prepare("SELECT u.*, a.id as app_id FROM `user` u JOIN application a ON u.application_id = a.id WHERE u.login = ?");

        $user = $stmt->fetch();

            sendResponse(['success' => false, 'error' => 'Неверный логин или пароль'], 401);
        }
        
        if ($user['application_id'] != $id) {
            sendResponse(['success' => false, 'error' => 'Нет доступа к этой заявке'], 403);

        $stmt->execute([$validated['full_name'], $validated['phone'], $validated['email'], $validated['birthdate'], $validated['gender'], $validated['biography'], $validated['contract_accepted'], $id]);
        
        $pdo->prepare("DELETE FROM application_language WHERE application_id = ?")->execute([$id]);
        $lang_stmt = $pdo->prepare("SELECT id FROM programming_language WHERE name = ?");
        $insert_stmt = $pdo->prepare("INSERT INTO application_language (application_id, language_id) VALUES (?, ?)");
        foreach ($validated['languages'] as $lang) {
            $lang_stmt->execute([$lang]);
            $lang_id = $lang_stmt->fetchColumn();
            if ($lang_id) {
                $insert_stmt->execute([$id, $lang_id]);
            }
        }
        
        sendResponse(['success' => true, 'message' => 'Данные обновлены!']);
    }
    
    sendResponse(['success' => false, 'error' => 'Метод не поддерживается'], 405);

document.addEventListener('DOMContentLoaded', function() {
    const form = document.getElementById('application-form');
    if (!form) {
        console.log('Форма с id="application-form" не найдена');
        return;
    }
    
    const apiUrl = form.dataset.apiUrl || 'api.php';
    const isLoggedIn = form.dataset.isLoggedIn === 'true';
    const applicationId = form.dataset.applicationId || null;
    const userLogin = form.dataset.userLogin || '';
    const userPassword = form.dataset.userPassword || '';
    
    form.addEventListener('submit', async function(e) {
        e.preventDefault();
        
        const formData = new FormData(form);
        const data = {};
        const languages = [];
        
        formData.forEach((value, key) => {
            if (key === 'languages[]' || key === 'languages') {
                languages.push(value);
            } else {
                data[key] = value;
            }
        });
        data.languages = languages;
        
        let method = 'POST';
        let url = apiUrl;
        
        if (isLoggedIn && applicationId) {
            method = 'PUT';
            url = `${apiUrl}?id=${applicationId}`;
        }
        
        const headers = {
            'Content-Type': 'application/json'
        };
        
        if (method === 'PUT' && userLogin && userPassword) {
            headers['Authorization'] = 'Basic ' + btoa(userLogin + ':' + userPassword);
        }
        
        try {
            const response = await fetch(url, {
                method: method,
                headers: headers,
                body: JSON.stringify(data)
            });
            
            const result = await response.json();
            
            const messageDiv = document.getElementById('api-message') || document.createElement('div');
            messageDiv.id = 'api-message';
            
            if (result.success) {
                messageDiv.className = 'success-message';
                messageDiv.innerHTML = `<strong>✅ ${result.message}</strong>`;
                if (result.login && result.password) {
                    messageDiv.innerHTML += `<br><strong>Логин:</strong> ${result.login}<br>
                                            <strong>Пароль:</strong> ${result.password}<br>
                                            <a href="${result.profile_url}" target="_blank">Перейти в профиль</a>`;
                }
                document.querySelectorAll('.field-error').forEach(el => el.remove());
                document.querySelectorAll('.error').forEach(el => el.classList.remove('error'));
            } else {
                messageDiv.className = 'error-message';
                const errors = result.errors || { global: [result.error] };
                messageDiv.innerHTML = `<strong>⚠️ Ошибки:</strong><ul>${Object.values(errors).flat().map(e => `<li>${e}</li>`).join('')}</ul>`;
                
                document.querySelectorAll('.field-error').forEach(el => el.remove());
                document.querySelectorAll('.error').forEach(el => el.classList.remove('error'));
                
                for (const [field, errorMsg] of Object.entries(errors)) {
                    if (field === 'global') continue;
                    const input = form.querySelector(`[name="${field}"]`) || form.querySelector(`[name="${field}[]"]`);
                    if (input) {
                        input.classList.add('error');
                        const errorDiv = document.createElement('div');
                        errorDiv.className = 'field-error';
                        errorDiv.textContent = errorMsg;
                        input.parentNode.insertBefore(errorDiv, input.nextSibling);
                    }
                }
            }
            
            if (!document.getElementById('api-message')) {
                form.insertAdjacentElement('beforebegin', messageDiv);
            }
        } catch (error) {
            console.error('Ошибка:', error);
        }
    });
});

$error = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $login = trim($_POST['login'] ?? '');
    $pass = $_POST['password'] ?? '';
    if ($login && $pass) {
        try {
            $stmt = $pdo->prepare("SELECT * FROM `user` WHERE login = ?");
            $stmt->execute([$login]);
            $user = $stmt->fetch();
            if ($user && password_verify($pass, $user['password_hash'])) {
                $_SESSION['user_id'] = $user['id'];
                header('Location: index.php');
                exit;
            } else {
                $error = 'Неверный логин или пароль';
            }
        } catch (PDOException $e) {
            $error = 'Ошибка БД';
