<?php
require_once 'config.php';

// Проверка CSRF-токена
if (!isset($_POST['csrf_token']) || $_POST['csrf_token'] !== $_SESSION['csrf_token']) {
    die('CSRF атака!');
}

$errors = [];
$old_data = [];
$valid_languages = ['Pascal', 'C', 'C++', 'JavaScript', 'PHP', 'Python', 'Java', 'Haskell', 'Clojure', 'Prolog', 'Scala', 'Go'];

function generateLogin($app_id) {
    return 'user_' . $app_id . '_' . bin2hex(random_bytes(4));
}
function generatePassword($len = 10) {
    return substr(str_shuffle('abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'), 0, $len);
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $full_name = trim($_POST['full_name'] ?? '');
    $phone = trim($_POST['phone'] ?? '');
    $email = trim($_POST['email'] ?? '');
    $birthdate = $_POST['birthdate'] ?? '';
    $gender = $_POST['gender'] ?? '';
    $languages = $_POST['languages'] ?? [];
    $biography = trim($_POST['biography'] ?? '');
    $contract_accepted = isset($_POST['contract_accepted']) ? 1 : 0;
    
    $old_data = compact('full_name', 'phone', 'email', 'birthdate', 'gender', 'languages', 'biography', 'contract_accepted');
    
    // Валидация
    if (empty($full_name)) {
        $errors['full_name'] = 'ФИО обязательно';
    } elseif (strlen($full_name) > 150) {
        $errors['full_name'] = 'ФИО не больше 150 символов';
    } elseif (!preg_match('/^[a-zA-Zа-яА-ЯёЁ\s\-]+$/u', $full_name)) {
        $errors['full_name'] = 'Только буквы, пробелы, дефисы';
    }
    
    if (empty($phone)) {
        $errors['phone'] = 'Телефон обязателен';
    } elseif (!preg_match('/^[\+\d\s\(\)\-]{10,20}$/', $phone)) {
        $errors['phone'] = 'Некорректный телефон';
    }
    
    if (empty($email)) {
        $errors['email'] = 'Email обязателен';
    } elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $errors['email'] = 'Некорректный email';
    }
    
    if (empty($birthdate)) {
        $errors['birthdate'] = 'Дата рождения обязательна';
    } elseif (!preg_match('/^\d{4}-\d{2}-\d{2}$/', $birthdate)) {
        $errors['birthdate'] = 'Формат ГГГГ-ММ-ДД';
    } else {
        $ts = strtotime($birthdate);
        if ($ts < strtotime('1900-01-01') || $ts > time()) {
            $errors['birthdate'] = 'Дата от 1900 до сегодня';
        }
    }
    
    if (!in_array($gender, ['male', 'female', 'other'])) {
        $errors['gender'] = 'Выберите пол';
    }
    
    if (empty($languages)) {
        $errors['languages'] = 'Выберите хотя бы один язык';
    } else {
        foreach ($languages as $lang) {
            if (!in_array($lang, $valid_languages)) {
                $errors['languages'] = "Недопустимый язык: $lang";
                break;
            }
        }
    }
    
    if (strlen($biography) > 5000) {
        $errors['biography'] = 'Биография не больше 5000 символов';
    }
    
    if (!$contract_accepted) {
        $errors['contract_accepted'] = 'Подтвердите контракт';
    }
    
    if (empty($errors)) {
        try {
            if (isset($_SESSION['user_id'])) {
                // UPDATE
                $user_id = $_SESSION['user_id'];
                $stmt = $pdo->prepare("SELECT application_id FROM `user` WHERE id = ?");
                $stmt->execute([$user_id]);
                $app_id = $stmt->fetchColumn();
                
                if ($app_id) {
                    $stmt = $pdo->prepare("UPDATE application SET full_name=?, phone=?, email=?, birthdate=?, gender=?, biography=?, contract_accepted=? WHERE id=?");
                    $stmt->execute([$full_name, $phone, $email, $birthdate, $gender, $biography, $contract_accepted, $app_id]);
                    
                    $pdo->prepare("DELETE FROM application_language WHERE application_id=?")->execute([$app_id]);
                    $lang_stmt = $pdo->prepare("SELECT id FROM programming_language WHERE name=?");
                    $insert_stmt = $pdo->prepare("INSERT INTO application_language (application_id, language_id) VALUES (?, ?)");
                    foreach ($languages as $lang) {
                        $lang_stmt->execute([$lang]);
                        $lang_id = $lang_stmt->fetchColumn();
                        if ($lang_id) $insert_stmt->execute([$app_id, $lang_id]);
                    }
                    header('Location: index.php?success=' . urlencode('Данные обновлены!'));
                    exit;
                }
            } else {
                // INSERT
                $pdo->beginTransaction();
                $stmt = $pdo->prepare("INSERT INTO application (full_name, phone, email, birthdate, gender, biography, contract_accepted) VALUES (?,?,?,?,?,?,?)");
                $stmt->execute([$full_name, $phone, $email, $birthdate, $gender, $biography, $contract_accepted]);
                $app_id = $pdo->lastInsertId();
                
                $lang_stmt = $pdo->prepare("SELECT id FROM programming_language WHERE name=?");
                $insert_stmt = $pdo->prepare("INSERT INTO application_language (application_id, language_id) VALUES (?, ?)");
                foreach ($languages as $lang) {
                    $lang_stmt->execute([$lang]);
                    $lang_id = $lang_stmt->fetchColumn();
                    if ($lang_id) $insert_stmt->execute([$app_id, $lang_id]);
                }
                
                $login = generateLogin($app_id);
                $plain_password = generatePassword();
                $hash = password_hash($plain_password, PASSWORD_DEFAULT);
                $stmt = $pdo->prepare("INSERT INTO `user` (login, password_hash, application_id) VALUES (?, ?, ?)");
                $stmt->execute([$login, $hash, $app_id]);
                
                $pdo->commit();
                
                $cookie_data = $old_data;
                setcookie('user_data', json_encode($cookie_data), time() + 365*24*3600, '/');
                
                header('Location: index.php?success=' . urlencode('Данные сохранены!') . '&login=' . urlencode($login) . '&password=' . urlencode($plain_password));
                exit;
            }
        } catch (PDOException $e) {
            if (isset($pdo) && $pdo->inTransaction()) $pdo->rollBack();
            $errors['global'][] = 'Ошибка БД';
        }
    }
    
    if (!empty($errors)) {
        setcookie('form_errors', json_encode($errors), time() + 3600, '/');
        header('Location: index.php?old=' . urlencode(json_encode($old_data)));
        exit;
    }
}
header('Location: index.php');
exit;

    }
}

if (!$auth_ok) {

}

$id = isset($_GET['id']) ? (int)$_GET['id'] : 0;
$error = '';
$application = null;
$selected_languages = [];
$languages = ['Pascal', 'C', 'C++', 'JavaScript', 'PHP', 'Python', 'Java', 'Haskell', 'Clojure', 'Prolog', 'Scala', 'Go'];

// Получаем данные заявки
try {
    $stmt = $pdo->prepare("SELECT * FROM application WHERE id = ?");
    $stmt->execute([$id]);
    $application = $stmt->fetch();
    
    if (!$application) {
        header('Location: admin.php?msg=' . urlencode('Заявка не найдена'));
        exit;
    }
    
    $stmt = $pdo->prepare("
        SELECT pl.name FROM application_language al
        JOIN programming_language pl ON al.language_id = pl.id
        WHERE al.application_id = ?
    ");
    $stmt->execute([$id]);
    $selected_languages = $stmt->fetchAll(PDO::FETCH_COLUMN);
    
} catch (PDOException $e) {
    $error = 'Ошибка БД';
}

// Обработка сохранения
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    // Проверка CSRF-токена
    if (!isset($_POST['csrf_token']) || $_POST['csrf_token'] !== $_SESSION['csrf_token']) {
        die('CSRF атака!');
    }
    
    $full_name = trim($_POST['full_name'] ?? '');
    $phone = trim($_POST['phone'] ?? '');
    $email = trim($_POST['email'] ?? '');
    $birthdate = $_POST['birthdate'] ?? '';
    $gender = $_POST['gender'] ?? '';
    $languages_post = $_POST['languages'] ?? [];
    $biography = trim($_POST['biography'] ?? '');
    $contract_accepted = isset($_POST['contract_accepted']) ? 1 : 0;
    
    $errors = [];
    
    // Валидация
    if (empty($full_name)) $errors[] = 'ФИО обязательно';
    elseif (strlen($full_name) > 150) $errors[] = 'ФИО не более 150 символов';
    elseif (!preg_match('/^[a-zA-Zа-яА-ЯёЁ\s\-]+$/u', $full_name)) $errors[] = 'ФИО: только буквы, пробелы, дефисы';
    
    if (empty($phone)) $errors[] = 'Телефон обязателен';
    elseif (!preg_match('/^[\+\d\s\(\)\-]{10,20}$/', $phone)) $errors[] = 'Некорректный телефон';
    
    if (empty($email)) $errors[] = 'Email обязателен';
    elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) $errors[] = 'Некорректный email';
    
    if (empty($birthdate)) $errors[] = 'Дата рождения обязательна';
    else {
        $ts = strtotime($birthdate);
        if ($ts < strtotime('1900-01-01') || $ts > time()) $errors[] = 'Дата от 1900 до сегодня';
    }
    
    if (!in_array($gender, ['male', 'female', 'other'])) $errors[] = 'Выберите пол';
    
    if (empty($languages_post)) $errors[] = 'Выберите хотя бы один язык';
    else {
        foreach ($languages_post as $lang) {
            if (!in_array($lang, $languages)) {
                $errors[] = "Недопустимый язык: $lang";
                break;
            }
        }
    }
    
    if (strlen($biography) > 5000) $errors[] = 'Биография не более 5000 символов';
    if (!$contract_accepted) $errors[] = 'Подтвердите контракт';
    
    if (empty($errors)) {
        try {
            $pdo->beginTransaction();
            
            $stmt = $pdo->prepare("
                UPDATE application SET 
                full_name = ?, phone = ?, email = ?, birthdate = ?, 
                gender = ?, biography = ?, contract_accepted = ?
                WHERE id = ?
            ");
            $stmt->execute([$full_name, $phone, $email, $birthdate, $gender, $biography, $contract_accepted, $id]);
            
            $pdo->prepare("DELETE FROM application_language WHERE application_id = ?")->execute([$id]);
            
            $lang_stmt = $pdo->prepare("SELECT id FROM programming_language WHERE name = ?");
            $insert_stmt = $pdo->prepare("INSERT INTO application_language (application_id, language_id) VALUES (?, ?)");
            
            foreach ($languages_post as $lang_name) {
                $lang_stmt->execute([$lang_name]);
                $lang_id = $lang_stmt->fetchColumn();
                if ($lang_id) $insert_stmt->execute([$id, $lang_id]);
            }
            
            $pdo->commit();
            
            header('Location: admin.php?msg=' . urlencode('Заявка #' . $id . ' обновлена'));
            exit;
            
        } catch (PDOException $e) {
            if ($pdo->inTransaction()) $pdo->rollBack();
            $error = 'Ошибка БД';
        }
    } else {
        $error = implode('<br>', $errors);
    }
}
